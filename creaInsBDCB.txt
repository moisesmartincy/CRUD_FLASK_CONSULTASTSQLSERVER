create database concursoBaile
use concursoBaile

create table ConcursoBaile
(
nroC integer not null,
fechaIni date,
fechaFin date,
primary key(nroC)
);

create table danza
(
codDan varchar(5)not null,
nombre varchar(100),
descr varchar(300),
primary key(codDan)
);

create table institucion
(
codIns varchar(5) not null,
nombre varchar(50),
descr varchar(100),
primary key(codIns)
);

create table participante
(
codPar varchar(5)not null,
nombre varchar(15),
paterno varchar(15),
materno varchar(15),
fecnac date,
sexo varchar(1),
tipo varchar(2), --participante, jurado, asesor
codIns varchar(5),
primary key(codPar),
foreign key(codIns) references Institucion
);

create table Musica
(
codMus varchar(5)not null,
titulo varchar(100),
autor varchar(50),
tipo varchar(10),
primary key(codMus)
);

create table baile
(
nroBai integer not null,
codParMuj varchar(5)not null,
codParVar varchar(5)not null,
fecha date,
horaIni varchar(5),
horaFin varchar(5),
ptjeTotal integer,
codDan varchar(5),
codCou varchar(5),
codMus varchar(5),
codCon integer,
primary key(nroBai),
foreign key(codParMuj) references Participante,
foreign key(codParVar) references Participante,
foreign key(codDan) references Danza,
foreign key(codCou) references Participante,
foreign key(codMus) references Musica,
foreign key(codCon) references ConcursoBaile
);

create table DetalleBaile
(
nroBai integer not null,
nroDet integer not null,
codJue varchar(5)not null,
ptje integer,
obs varchar(100),
primary key(nroBai, nroDet),
foreign key(codJue) references Participante
);

create table Auspiciador
(
codAus varchar(5) not null,
nombre varchar(50) not null,
descr varchar(50),
primary key(codAus)
);

create table Publicidad
(
nroCon integer not null,
codAus varchar(5) not null,
fechaIni varchar(50) not null,
producto varchar(50),
primary key(nroCon,codAus),
foreign key(nroCon) references ConcursoBaile,
foreign key(codAus) references Auspiciador
);
